	x = 0
	y = 1
	cost = 5

	available = {
		OR = {
			tag = ITA
			tag = GER
			tag = TUR
		}
	}

	completion_reward = {
		if = {
			limit = {
				is_in_faction = no
			}
			set_rule = { can_create_factions = yes }
			create_faction = "I Nuovi Zappatori"
		}
		else = {
			set_faction_name = "I Nuovi Zappatori"
		}

		if = {
			limit = {
				NOT = { tag = ITA }
				country_exists = ITA
			}
			ITA = {
				country_event = { id = zappatori_event.1 days = 1 }
			}
		}
		if = {
			limit = {
				NOT = { tag = GER }
				country_exists = GER
			}
			GER = {
				country_event = { id = zappatori_event.1 days = 1 }
			}
		}
		if = {
			limit = {
				NOT = { tag = TUR }
				country_exists = TUR
			}
			TUR = {
				country_event = { id = zappatori_event.1 days = 1 }
			}
		}
	}
}
